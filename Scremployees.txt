      OnVisible: |-
        =Set(varLoading, true);

        Refresh(SkillLevels);
        Refresh(EmployeeSkillsHistory);

        ClearCollect(colSkills, SkillsList);
        ClearCollect(colLevels, SkillLevels);
        Clear(colHistory);
        ForAll(
            colSkills As s,
            Collect(colHistory, Filter(EmployeeSkillsHistory, SkillName = s.SkillName))
        );
        ClearCollect(
            colLatest,
            Sort(
                colHistory,
                Coalesce(DateDiff(Date(2000, 1, 1), UpdatedDate), 0) * 100000
                    + Coalesce(Value(Mid(RecordID, 5)), 0),
                SortOrder.Descending
            )
        );
